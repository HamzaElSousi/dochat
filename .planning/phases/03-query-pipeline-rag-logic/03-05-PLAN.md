import json
import uuid
import pytest
import struct
from unittest.mock import MagicMock, call


FAKE_EMBEDDING = [0.1] * 1536
FAKE_ANSWER = "This is a test answer from the LLM."


def _pack_f32(vec):
    return struct.pack(f"{len(vec)}f", *vec)


@pytest.fixture
def seeded_db(app):
    """Insert one document + one chunk + one vec_items row into the test DB.
    Returns (chunk_id, doc_id, filename) for test assertions.
    """
    conn = app.config['DB_CONN']
    doc_id = str(uuid.uuid4())
    chunk_id = str(uuid.uuid4())
    now = "2026-01-01T00:00:00+00:00"

    conn.execute("BEGIN")
    conn.execute(
        "INSERT INTO documents (id, filename, filetype, uploaded_at, status, chunk_count, filepath) "
        "VALUES (?, 'test.txt', 'txt', ?, 'ready', 1, NULL)",
        [doc_id, now],
    )
    conn.execute(
        "INSERT INTO chunks (id, doc_id, content, chunk_index) VALUES (?, ?, 'Test chunk content', 0)",
        [chunk_id, doc_id],
    )
    result = conn.execute(
        "INSERT INTO vec_items(embedding) VALUES (?)",
        [_pack_f32(FAKE_EMBEDDING)],
    )
    vec_rowid = result.lastrowid
    conn.execute(
        "INSERT INTO chunk_embeddings (chunk_id, vec_rowid) VALUES (?, ?)",
        [chunk_id, vec_rowid],
    )
    conn.execute("COMMIT")
    return {"chunk_id": chunk_id, "doc_id": doc_id, "filename": "test.txt"}

def test_chat_multiturn_session(client, seeded_db, mocker):
    """Two turns with same session_id → second LLM call includes prior turn in messages (QUERY-04)."""
    mocker.patch('app.services.query.embed_query', return_value=FAKE_EMBEDDING)
    mock_llm = mocker.patch('app.services.query._call_llm', return_value=FAKE_ANSWER)
    mocker.patch('app.services.query._vector_search',
                 return_value=[(seeded_db['chunk_id'], 0.2)])

    # First turn — get session_id
    r1 = client.post('/chat', json={'message': 'First question'})
    assert r1.status_code == 200
    session_id = r1.get_json()['session_id']

    # Second turn — use same session_id
    r2 = client.post('/chat', json={'message': 'Follow-up question', 'session_id': session_id})
    assert r2.status_code == 200

    # The second _call_llm invocation should include prior conversation in messages
    assert mock_llm.call_count == 2
    second_call_messages = mock_llm.call_args_list[1][0][0]  # first positional arg = messages list
    # System prompt is always first; history turns appear before the current user message
    message_roles = [m['role'] for m in second_call_messages]
    assert 'system' in message_roles
    # The prior user turn should appear in messages
    message_contents = [m['content'] for m in second_call_messages]
    assert any('First question' in c for c in message_contents)

def test_chat_history_trimming(client, seeded_db, monkeypatch, mocker):
    """After 11 turns, only last 10 sent to LLM (QUERY-04 — MAX_HISTORY_TURNS=10)."""
    mocker.patch('app.services.query.embed_query', return_value=FAKE_EMBEDDING)
    mock_llm = mocker.patch('app.services.query._call_llm', return_value=FAKE_ANSWER)
    mocker.patch('app.services.query._vector_search',
                 return_value=[(seeded_db['chunk_id'], 0.2)])

    # Send 11 turns
    session_id = None
    for i in range(11):
        r = client.post('/chat', json={
            'message': f'Question {i}',
            **({'session_id': session_id} if session_id else {}),
        })
        assert r.status_code == 200
        session_id = r.get_json()['session_id']

    # 12th turn — LLM messages should contain at most 10 prior turns (20 messages)
    # plus system prompt + current user message = 22 messages max
    last_call_messages = mock_llm.call_args_list[-1][0][0]
    # Filter out system prompt and current user message to count history
    history_messages = [m for m in last_call_messages
                        if m['role'] != 'system']
    # History (trimmed) + current user message: max 10*2 + 1 = 21
    # Current user message is the last one; all prior are history
    assert len(history_messages) <= 21  # 10 turns * 2 messages + 1 current
